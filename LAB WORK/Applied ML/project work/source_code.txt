import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.impute import SimpleImputer

#importing dataset
df=pd.read_csv('Life.csv')
print(df.info())


# Select numeric features 
numeric_features = df.select_dtypes(include=np.number)

# Impute missing values in numeric features
imputer = SimpleImputer(strategy='mean')
imputed_data = pd.DataFrame(
    imputer.fit_transform(numeric_features),
    columns=numeric_features.columns
)

# Combine: imputed numeric data + Country column
df = pd.concat([imputed_data, df[['Country']]], axis=1)

# Check remaining null values
df.isnull().sum()


#data cleaning
india= df[df['Country'].str.strip()=='India'].sort_values('Year')
india[['Year','infant deaths','Adult Mortality','Alcohol']].head(15)


#data analysis
india[['Year','infant deaths','Adult Mortality','Alcohol']].describe()

#Visual representation and analysis(for Year and infant mortality)
x_year= india['Year'].values
y_infant_deaths= india['infant deaths'].values
plt.scatter(x_year, y_infant_deaths, color= 'red')
plt.xlabel('Years')
plt.ylabel('Infant Deaths')
plt.title('Infant Mortality vs Year(India)')
plt.show()

#data exploration
x_year = x_year.reshape(-1,1)
x_year

#traning model
model= LinearRegression().fit(x_year, y_infant_deaths)
model

#model predictions
prediction= model.predict(x_year)
prediction

#Data representation
plt.scatter(x_year,y_infant_deaths, color='red')
plt.plot(x_year,prediction, color='green')
plt.xlabel('Years')
plt.ylabel('Infant Deaths')
plt.title('Model Predictions without train test split(India)')
plt.show()


#splitting data into training and testing(for year and infant mortality)
x_train, x_test, y_train, y_test = train_test_split(
    x_year, y_infant_deaths, train_size=0.7, test_size=0.3, random_state=42
)
print(f"X_train shape ={x_train.shape}")
print(f"Y_train shape ={y_train.shape}")
print(f"X_test shape ={x_test.shape}")
print(f"Y_test shape ={x_test.shape}")

# data representation for training data(for year and infant mortality)
plt.scatter(x_train,y_train, color='green')
plt.xlabel('Years')
plt.ylabel('Infant Deaths')
plt.title('Training data')
plt.show()

# data depresentation for testing data(for year and infant mortality)
plt.scatter(x_test,y_test, color='red')
plt.xlabel('Years')
plt.ylabel('Infant Deaths')
plt.title('Testing data')
plt.show()

#training the model(for infant deaths and year)
model=LinearRegression()
model.fit(x_train,y_train)


#model predictions
y_pred=model.predict(x_test)
y_pred

# 1. Weight (Slope)
print("Weight (Slope) for Infant Mortality Prediction:", model.coef_)

# 2. Bias (Intercept)
print("Bias (Intercept) for Infant Mortality Prediction:", model.intercept_)


from sklearn.metrics import mean_absolute_error,mean_squared_error
from sklearn.metrics import r2_score
# 1. Mean Squared Error (MSE)
mse_inf = mean_squared_error(y_test, y_pred)
print("Mean Squared Error (MSE) for Infant Mortality Prediction:", mse_inf)

# 2. Mean Absolute Error (MAE)
mae_inf = mean_absolute_error(y_test, y_pred)
print("Mean Absolute Error (MAE) for Infant Mortality Prediction:", mae_inf)

# 2. R-square Error
r2_inf= r2_score(y_test, y_pred)
print("R² Score for Infant Mortality Prediction:", r2_inf)


#Visual representation for Infant Mortality prediction
plt.scatter(x_test,y_test, color= 'red')
plt.plot(x_test, model.predict(x_test), color='blue')
plt.title('Infant Mortality vs Years (India)')
plt.xlabel('Years')
plt.ylabel('Infant Deaths')
plt.show()
